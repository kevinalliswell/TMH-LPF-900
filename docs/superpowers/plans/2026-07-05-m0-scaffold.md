/// CRC-16/MODBUS checksum over `data` (poly 0x8005, init 0xFFFF, reflected).
///
/// Returned as a `u16`; Modbus-RTU frames carry it low-byte-first on the wire.
pub fn crc16_modbus(data: &[u8]) -> u16 {
    const MODBUS: Crc<u16> = Crc::<u16>::new(&CRC_16_MODBUS);
    MODBUS.checksum(data)
}

- [ ] **Step 6: 写占位图标生成脚本 `scripts/gen_icon.py`**

```python
#!/usr/bin/env python3
"""Generate a 1024x1024 solid dark-slate RGBA PNG placeholder icon.
Dependency-free (stdlib zlib/struct). Usage: python3 scripts/gen_icon.py <out.png>
"""
import struct
import sys
import zlib

W = H = 1024
PIXEL = bytes((11, 15, 20, 255))  # #0b0f14 opaque, matches app dark bg


def chunk(tag: bytes, data: bytes) -> bytes:
    body = tag + data
    return struct.pack(">I", len(data)) + body + struct.pack(">I", zlib.crc32(body) & 0xFFFFFFFF)


def main() -> None:
    out = sys.argv[1]
    row = b"\x00" + PIXEL * W  # filter byte 0 (None) + one row of pixels
    raw = row * H
    png = b"\x89PNG\r\n\x1a\n"
    png += chunk(b"IHDR", struct.pack(">IIBBBBB", W, H, 8, 6, 0, 0, 0))  # RGBA, 8-bit
    png += chunk(b"IDAT", zlib.compress(raw, 9))
    png += chunk(b"IEND", b"")
    with open(out, "wb") as f:
        f.write(png)
    print(f"wrote {out} ({W}x{H})")


if __name__ == "__main__":
    main()
```

## Task 5: 版本管理(单一来源同步 + CHANGELOG)

- [ ] **Step 1: 写 `scripts/sync-version.mjs`**

```js
#!/usr/bin/env node
// Single-source version sync: package.json.version -> the 3 display points
// (tauri.conf.json version + window title, index.html <title>, App.tsx splash).
//   node scripts/sync-version.mjs           write mode (updates files)
//   node scripts/sync-version.mjs --check    verify mode (exit 1 on drift)
import { readFileSync, writeFileSync } from "node:fs";
import { fileURLToPath } from "node:url";
import { dirname, join } from "node:path";

const root = join(dirname(fileURLToPath(import.meta.url)), "..");
const check = process.argv.includes("--check");
const pkg = JSON.parse(readFileSync(join(root, "package.json"), "utf8"));
const version = pkg.version;
const title = `TMH-LPF-900 v${version}`;

const targets = [
  {
    file: "src-tauri/tauri.conf.json",
    transform: (s) => {
      const c = JSON.parse(s);
      c.version = version;
      c.app.windows[0].title = title;
      return JSON.stringify(c, null, 2) + "\n";
    },
  },
  {
    file: "index.html",
    transform: (s) => s.replace(/<title>[^<]*<\/title>/, `<title>${title}</title>`),
  },
  {
    file: "src/App.tsx",
    transform: (s) => s.replace(/TMH-LPF-900 v\d+\.\d+\.\d+/g, title),
  },
];

let drift = false;
for (const { file, transform } of targets) {
  const path = join(root, file);
  const cur = readFileSync(path, "utf8");
  const next = transform(cur);
  if (cur === next) {
    console.log(`ok       ${file}`);
  } else if (check) {
    console.error(`MISMATCH ${file} (expected ${version})`);
    drift = true;
  } else {
    writeFileSync(path, next);
    console.log(`updated  ${file}`);
  }
}

if (check && drift) {
  console.error("\nVersion drift. Run: pnpm sync-version");
  process.exit(1);
}
console.log(`\nAll display points at version ${version}.`);
```

```yaml
name: CI

on:

concurrency:
  group: ci-${{ github.ref }}

          components: clippy, rustfmt

      - uses: Swatinem/rust-cache@v2

      - name: Install pnpm
        uses: pnpm/action-setup@v4

      - name: Install Node
        uses: actions/setup-node@v4
        with:
          node-version-file: .nvmrc
          cache: pnpm

      - name: Install Linux Tauri system deps
        if: matrix.os == 'ubuntu-latest'
        run: |
          sudo apt-get update
          sudo apt-get install -y \
            libwebkit2gtk-4.1-dev \
            libgtk-3-dev \
            librsvg2-dev \
            libayatana-appindicator3-dev \
            patchelf

      - name: cargo fmt
        run: cargo fmt --all --check

      - name: pnpm install
        run: pnpm install --frozen-lockfile

      - name: Frontend typecheck / lint / test
        run: pnpm typecheck && pnpm lint && pnpm test

      - name: Frontend build (produces dist/ for tauri context)
        run: pnpm build

      - name: cargo clippy
        run: cargo clippy --workspace --all-targets -- -D warnings

      - name: cargo test
        run: cargo test --workspace

      - name: Version consistency
        run: node scripts/sync-version.mjs --check

      - name: Dependency-direction guard (tmh-core / tmh-protocols must not depend on tauri)
        shell: bash
        run: |
          set -euo pipefail
          fail=0
          for crate in tmh-core tmh-protocols; do
            if cargo tree -p "$crate" -e normal,build --prefix none | grep -qiE '^tauri'; then
              echo "::error::$crate must not depend on tauri"
              cargo tree -p "$crate" -e normal,build
              fail=1
            else
              echo "ok: $crate has no tauri dependency"
            fi
          done
          exit $fail
```

- [ ] **Step 1: 写 `.github/workflows/release.yml`**

```yaml
name: Release

on:
  push:
    tags: ["v*"]
  workflow_dispatch:

jobs:
  verify-version:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Tag must match package.json version
        if: startsWith(github.ref, 'refs/tags/v')
        shell: bash
        run: |
          set -euo pipefail
          TAG="${GITHUB_REF#refs/tags/v}"
          PKG="$(node -p "require('./package.json').version")"
          echo "tag=$TAG package.json=$PKG"
          if [ "$TAG" != "$PKG" ]; then
            echo "::error::tag v$TAG != package.json $PKG"; exit 1
          fi

  release:
    needs: verify-version
    permissions:
      contents: write
    strategy:
      fail-fast: false
      matrix:
        include:
          - os: macos-latest
            args: "--target universal-apple-darwin"
            targets: "aarch64-apple-darwin,x86_64-apple-darwin"
          - os: windows-latest
            args: ""
            targets: ""
    runs-on: ${{ matrix.os }}
    steps:
      - uses: actions/checkout@v4

      - name: Install Rust (stable)
        uses: dtolnay/rust-toolchain@stable
        with:
          targets: ${{ matrix.targets }}

      - uses: Swatinem/rust-cache@v2

      - name: Install pnpm
        uses: pnpm/action-setup@v4

      - name: Install Node
        uses: actions/setup-node@v4
        with:
          node-version-file: .nvmrc
          cache: pnpm

      - name: pnpm install
        run: pnpm install --frozen-lockfile

      - name: Build bundles & draft release
        uses: tauri-apps/tauri-action@v0
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
        with:
          tagName: ${{ github.ref_type == 'tag' && github.ref_name || format('v0.0.0-dev.{0}', github.run_number) }}
          releaseName: TMH-LPF-900 ${{ github.ref_type == 'tag' && github.ref_name || 'dev build' }}
          releaseBody: "Unsigned build. See CHANGELOG.md. Signing/updater land in M5."
          releaseDraft: true
          prerelease: false
          args: ${{ matrix.args }}
```
